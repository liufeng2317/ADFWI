[docs]def tensor2numpy(a):
    """transform tensor data into numpy
    """
    if not torch.is_tensor(a):
        return a 
    else:
        return a.detach().numpy()

    

[docs]def gpu2cpu(a):
    if torch.is_tensor(a):
        if a.requires_grad:
            if a.device == 'cpu':
                a = a.detach().numpy()
            else:
                a = a.cpu().detach().numpy()
        else:
            if a.device == 'cpu':
                return a.numpy()
            else:
                return a.cpu().numpy()
    return a
